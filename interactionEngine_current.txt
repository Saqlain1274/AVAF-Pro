"""
===============================================================================
AVAF PRO

Behavioral Interaction Engine
===============================================================================

Phase-1 browser-driven discovery.

Purpose
-------
Explore the REAL application state through Playwright.

The engine does not manufacture attack vectors.

It observes application behavior produced by:
    - rendered DOM
    - visible links
    - buttons
    - menus
    - dialogs
    - tabs
    - accordions
    - pagination controls
    - search/filter controls
    - editable controls
    - SPA/client-side navigation
    - browser events
    - runtime observations
    - DOM mutations

Every observation is produced from the browser state actually reached.

Phase-2 receives these observations and performs validation.

Safety boundary
---------------
This component performs discovery only.

It does NOT:
    - inject vulnerability payloads
    - brute force
    - guess URLs
    - guess parameter names
    - fuzz endpoints
    - submit destructive actions
    - purchase/pay/transfer/delete
    - upload arbitrary files
    - bypass authentication

Important
---------
The crawler URL QueueManager is NOT used as the interaction queue.

InteractionEngine owns its own browser-interaction queue.

===============================================================================
"""

from __future__ import annotations

import asyncio
from asyncio.log import logger
import hashlib
import heapq
import json
from dataclasses import dataclass, field
from typing import Any


# =============================================================================
# INTERACTION SURFACE
# =============================================================================

@dataclass(slots=True)
class InteractionSurface:

    priority: int

    action: str

    locator: str

    source: str = "rendered-dom"

    metadata: dict[str, Any] = field(
        default_factory=dict
    )

    visited: bool = False


# =============================================================================
# RESULT
# =============================================================================

def _empty_result() -> dict[str, Any]:

    return {
        "events": [],
        "navigations": [],
        "routes": [],
        "controls": [],
        "forms": [],
        "surfaces": [],
        "dom_states": [],
        "runtime_observations": [],
        "network_observations": [],
        "stats": {},
    }


# =============================================================================
# ENGINE
# =============================================================================

class InteractionEngine:

    def __init__(
        self,
        queue=None,
    ):

               # ------------------------------------------------------------------
        # Adaptive behavioral exploration
        # ------------------------------------------------------------------
        #
        # There is deliberately NO fixed number of pages, routes,
        # parameters, or findings.
        #
        # The engine follows real browser-observed state.
        #
        # max_depth / max_pages belong to PlaywrightCrawler and remain
        # the authoritative crawl-scope controls.
        #
        # InteractionEngine stops when the current browser state has
        # converged and there is no unexecuted surface that can produce
        # new evidence.
        # ------------------------------------------------------------------

        self.action_timeout = 7000

        self.state_wait = 500

        self.post_navigation_wait = 900

        # Number of consecutive genuinely stable observations required
        # before behavioral exploration is considered converged.
        #
        # This is NOT a finding/page/action limit.
        self.convergence_threshold = 2

        self._no_change_rounds = 0

        # Last observed browser state.
        self._last_state_fingerprint = ""

        # Fingerprints for states that have already been observed.
        self._visited_states = set()
        
        # ------------------------------------------------------------------
        # Compatibility reference to crawler queue.
        #
        # IMPORTANT:
        # We NEVER consume it for interaction surfaces.
        # ------------------------------------------------------------------

        self._crawler_queue = queue

        # ------------------------------------------------------------------
        # Runtime callback
        # ------------------------------------------------------------------

        self.runtime_callback = None

        # ------------------------------------------------------------------
        # Interaction priority queue
        # ------------------------------------------------------------------

        self._interaction_queue: list = []

        self._interaction_counter = 0

        # ------------------------------------------------------------------
        # Deduplication
        # ------------------------------------------------------------------

        self._seen_surfaces: set = set()

        self._executed: set = set()

        self._visited_states: set = set()

        self._observed_urls: set = set()

        self._observed_network: set = set()

        # ------------------------------------------------------------------
        # Browser state graph
        # ------------------------------------------------------------------

        self._graph: dict[str, dict[str, Any]] = {}

        # ------------------------------------------------------------------
        # Statistics
        # ------------------------------------------------------------------

        self.stats = {

            "rounds": 0,

            "actions": 0,

            "surfaces": 0,

            "clicks": 0,

            "hovers": 0,

            "forms": 0,

            "controls_filled": 0,

            "dialogs": 0,

            "navigation_changes": 0,

            "runtime_feedback": 0,

            "failed_actions": 0,

            "skipped_actions": 0,

            "dom_states": 0,

            "new_dom_states": 0,

            "new_routes": 0,

            "new_controls": 0,

            "new_forms": 0,

            "runtime_events": 0,

            "network_observations": 0,
        }

        self._result = _empty_result()

    # =========================================================================
    # CALLBACK
    # =========================================================================

    def set_runtime_callback(
        self,
        callback,
    ) -> None:

        self.runtime_callback = callback

    # =========================================================================
    # RESET
    # =========================================================================

    def reset(self):
        """
        Reset only InteractionEngine state.

        The crawler QueueManager is owned by PlaywrightCrawler and must
        NEVER be cleared here.
        """

        self._interaction_queue.clear()

        self._interaction_counter = 0

        self._seen_surfaces.clear()

        self._executed.clear()

        self._visited_states.clear()

        self._observed_urls.clear()

        self._observed_network.clear()

        self._graph.clear()

        self._no_change_rounds = 0

        self._last_state_fingerprint = ""

        self._result = _empty_result()

        for key in self.stats:
            self.stats[key] = 0
            
    # =========================================================================
    # NORMALIZATION HELPERS
    # =========================================================================

    @staticmethod
    def _as_list(
        value,
    ) -> list:
        """
        Convert any supported discovery value into a safe list.

        This specifically prevents:
            TypeError: 'NoneType' object is not iterable
        """

        if value is None:
            return []

        if isinstance(
            value,
            list,
        ):
            return value

        if isinstance(
            value,
            tuple,
        ):
            return list(value)

        if isinstance(
            value,
            set,
        ):
            return list(value)

        if isinstance(
            value,
            dict,
        ):
            return [value]

        return [value]

    @staticmethod
    def _safe_string(
        value,
    ) -> str:

        if value is None:
            return ""

        try:
            return str(value).strip()
        except Exception:
            return ""

    # =========================================================================
    # QUEUE
    # =========================================================================

    async def add_surface(
        self,
        surface: InteractionSurface,
    ):
        """
        Add one interaction opportunity belonging to the CURRENT
        rendered browser state.

        IMPORTANT
        ---------
        The same DOM locator may legitimately represent a different
        control after:

            click
                ↓
            SPA mutation
                ↓
            modal/menu/tab appears
                ↓
            DOM changes

        Therefore the browser-state fingerprint is part of the
        deduplication identity.

        This does NOT create routes or parameters.
        It only prevents the same observed control from being executed
        repeatedly inside the same rendered state.
        """

        state = str(
            surface.metadata.get(
                "state",
                "",
            )
        )

        identity = surface.metadata.get(
            "identity",
            {},
        )

        state_key = (
            surface.metadata or {}
        ).get(
            "state",
            "",
        )

        identity_key = (
            surface.metadata or {}
        ).get(
            "identity_hash",
            "",
        )

        key = (
            state_key,
            surface.action,
            surface.locator,
            identity_key,
        )

        if key in self._seen_surfaces:
            return

        self._seen_surfaces.add(key)

        self.stats["surfaces"] += 1

        self._interaction_counter += 1

        heapq.heappush(
            self._interaction_queue,
            (
                -int(surface.priority),
                self._interaction_counter,
                surface,
            ),
        )

    @staticmethod
    def _stable_metadata_key(
        metadata,
    ) -> str:

        try:

            return json.dumps(
                metadata or {},
                sort_keys=True,
                default=str,
            )

        except Exception:

            return repr(metadata)

    async def next_surface(
        self,
    ):

        if not self._interaction_queue:
            return None

        _, _, surface = heapq.heappop(
            self._interaction_queue
        )

        return surface

    def has_work(
        self,
    ) -> bool:

        return bool(
            self._interaction_queue
        )

    def mark_executed(
        self,
        surface,
    ) -> None:

        if surface is None:
            return

        metadata = (
            surface.metadata
            if isinstance(
                surface.metadata,
                dict,
            )
            else {}
        )

        state_key = metadata.get(
            "state",
            "",
        )

        identity_key = metadata.get(
            "identity_hash",
            "",
        )

        key = (
            state_key,
            surface.action,
            surface.locator,
            identity_key,
        )

        if key in self._executed:
            return

        self._executed.add(
            key
        )

        self.stats[
            "actions"
        ] += 1
        """
        Mark an observed interaction as executed.

        Execution identity includes the browser state so that an
        identical locator appearing in a newly rendered state is
        considered a new observation.
        """

        state = str(
            surface.metadata.get(
                "state",
                "",
            )
        )

        identity = surface.metadata.get(
            "identity",
            {},
        )

        key = (
            state,
            surface.action,
            surface.locator,
            repr(identity),
        )

        self._executed.add(key)

        self.stats["actions"] += 1
        
    def _evidence_count(
        self,
    ) -> int:
        """
        Return the amount of evidence currently held by the engine.

        This is used only to determine whether an interaction produced
        something new.

        It does NOT represent a vulnerability count.
        """

        return (
            len(
                self._visited_states
            )
            +
            len(
                self._observed_urls
            )
            +
            len(
                self._observed_network
            )
            +
            len(
                self._result.get(
                    "events",
                    [],
                )
            )
            +
            len(
                self._result.get(
                    "controls",
                    [],
                )
            )
            +
            len(
                self._result.get(
                    "forms",
                    [],
                )
            )
        )

    # =========================================================================
    # MAIN EXPLORATION
    # =========================================================================
    
    async def explore(
        self,
        page,
        decision_engine=None,
    ) -> dict:
        """
        Adaptive browser-driven behavioral discovery.

        IMPORTANT
        ---------
        This engine has no predefined number of:
            - pages
            - routes
            - parameters
            - actions
            - exploration rounds

        The crawler owns crawl-scope limits such as max_pages/max_depth.

        InteractionEngine continues while real browser interaction produces
        new observable evidence or exposes new interaction surfaces.

        It stops only when the current browser state has genuinely
        converged and no unexecuted observed surface remains.

        Discovery only:
            - no URL guessing
            - no wordlists
            - no parameter guessing
            - no payload injection
            - no vulnerability validation
        """

        self.reset()

        try:
            from phase1_scanning_detection.discovery.logger import (
                logger as discovery_logger,
            )
        except Exception:
            discovery_logger = None

        def log_info(message):
            if discovery_logger:
                discovery_logger.info(
                    "INTERACTION",
                    message,
                )

        def log_debug(message):
            if discovery_logger:
                discovery_logger.debug(
                    "INTERACTION",
                    message,
                )

        if page is None:
            return self.export_graph()

        try:
            if page.is_closed():
                return self.export_graph()
        except Exception:
            return self.export_graph()

        current_url = self._safe_page_url(
            page
        )

        if not current_url:
            return self.export_graph()

        if current_url.startswith(
            (
                "about:blank",
                "chrome-error://",
                "chrome://",
            )
        ):
            return self.export_graph()

        log_info(
            (
                "Starting adaptive browser behavioral "
                f"exploration | url={current_url}"
            )
        )

        # -------------------------------------------------------------
        # Initial observation
        # -------------------------------------------------------------

        initial_state = await self._state_fingerprint(
            page
        )

        if initial_state:

            self._visited_states.add(
                initial_state
            )

            self._last_state_fingerprint = (
                initial_state
            )

            self.stats[
                "dom_states"
            ] += 1

            self.stats[
                "new_dom_states"
            ] += 1

            self._result[
                "dom_states"
            ].append(
                {
                    "fingerprint":
                        initial_state,

                    "url":
                        current_url,
                }
            )

        # -------------------------------------------------------------
        # Adaptive exploration
        # -------------------------------------------------------------

        while True:

            try:
                if page.is_closed():
                    break
            except Exception:
                break

            self.stats[
                "rounds"
            ] += 1

            round_number = (
                self.stats["rounds"]
            )

            log_info(
                (
                    "Behavioral observation cycle "
                    f"{round_number}"
                )
            )

            evidence_before = (
                self._evidence_count()
            )

            state_before = (
                await self._state_fingerprint(
                    page
                )
            )

            url_before = (
                self._safe_page_url(
                    page
                )
            )

            # =========================================================
            # 1. OBSERVE CURRENT LIVE DOM
            # =========================================================

            try:

                await self.discover_live_dom_surfaces(
                    page
                )

            except Exception as exc:

                log_debug(
                    (
                        "Live DOM discovery failed: "
                        f"{type(exc).__name__}: {exc}"
                    )
                )

            # =========================================================
            # 2. OBSERVE RUNTIME
            # =========================================================

            try:

                artifacts = (
                    await self._read_runtime_artifacts(
                        page
                    )
                )

            except Exception as exc:

                artifacts = []

                log_debug(
                    (
                        "Runtime artifact observation failed: "
                        f"{type(exc).__name__}: {exc}"
                    )
                )

            if not isinstance(
                artifacts,
                (list, tuple, set),
            ):
                artifacts = []

            # Runtime artifacts are evidence.
            # They may expose real application behavior but are NOT
            # converted into guessed URLs or guessed parameters.

            try:

                await self.discover_surfaces(
                    page,
                    list(artifacts),
                )

            except Exception as exc:

                log_debug(
                    (
                        "Runtime surface discovery failed: "
                        f"{type(exc).__name__}: {exc}"
                    )
                )

            # =========================================================
            # 3. CHECK WHETHER A REAL OBSERVED SURFACE EXISTS
            # =========================================================

            surface = await self.next_surface()

            if surface is None:

                # One final observation is important because runtime
                # callbacks/network activity can arrive after the DOM
                # queue temporarily becomes empty.

                await self._settle(
                    page,
                    self.state_wait,
                )

                try:
                    await self.discover_live_dom_surfaces(
                        page
                    )
                except Exception:
                    pass

                try:

                    runtime = (
                        await self._read_runtime_export(
                            page
                        )
                    )

                    self._record_runtime_observation(
                        runtime
                    )

                except Exception:
                    runtime = {}

                evidence_after = (
                    self._evidence_count()
                )

                state_after = (
                    await self._state_fingerprint(
                        page
                    )
                )

                new_evidence = (
                    evidence_after
                    > evidence_before
                )

                state_changed = (
                    state_after
                    and
                    state_after != state_before
                )

                if (
                    new_evidence
                    or
                    state_changed
                ):

                    self._no_change_rounds = 0

                    log_info(
                        (
                            "New browser evidence arrived "
                            "while interaction queue was empty; "
                            "continuing observation."
                        )
                    )

                    continue

                self._no_change_rounds += 1

                log_info(
                    (
                        "No unexecuted observed surface and "
                        "no new browser evidence | "
                        f"stable={self._no_change_rounds}/"
                        f"{self.convergence_threshold}"
                    )
                )

                if (
                    self._no_change_rounds
                    >= self.convergence_threshold
                ):
                    break

                continue

            # =========================================================
            # 4. REAL BROWSER INTERACTION
            # =========================================================

            action_success = False

            try:

                await self._execute_surface(
                    page,
                    surface,
                )

                action_success = True

            except Exception as exc:

                self.stats[
                    "failed_actions"
                ] += 1

                log_debug(
                    (
                        "Observed browser action failed | "
                        f"action={getattr(surface, 'action', '')} | "
                        f"locator={getattr(surface, 'locator', '')} | "
                        f"{type(exc).__name__}: {exc}"
                    )
                )

            # Always mark the observed surface as consumed.
            #
            # This prevents a single unchanged DOM control from being
            # executed forever.

            self.mark_executed(
                surface
            )

            if not action_success:

                self.stats[
                    "skipped_actions"
                ] += 1

            # =========================================================
            # 5. WAIT FOR REAL BROWSER/SPA STATE
            # =========================================================

            await self._settle_after_action(
                page,
                url_before,
            )

            # =========================================================
            # 6. COLLECT POST-ACTION RUNTIME
            # =========================================================

            try:

                await self._collect_post_action_runtime(
                    page
                )

            except Exception as exc:

                log_debug(
                    (
                        "Post-action runtime collection failed: "
                        f"{type(exc).__name__}: {exc}"
                    )
                )

            # =========================================================
            # 7. OBSERVE RESULTING DOM
            # =========================================================

            try:

                await self.discover_live_dom_surfaces(
                    page
                )

            except Exception as exc:

                log_debug(
                    (
                        "Post-action DOM discovery failed: "
                        f"{type(exc).__name__}: {exc}"
                    )
                )

            # =========================================================
            # 8. RECORD REAL STATE CHANGE
            # =========================================================

            state_after = (
                await self._state_fingerprint(
                    page
                )
            )

            url_after = (
                self._safe_page_url(
                    page
                )
            )

            evidence_after = (
                self._evidence_count()
            )

            state_changed = (
                state_after
                and
                state_after != state_before
            )

            url_changed = (
                url_after
                and
                url_after != url_before
            )

            evidence_changed = (
                evidence_after
                > evidence_before
            )

            if state_after:

                self.stats[
                    "dom_states"
                ] += 1

                if state_after not in self._visited_states:

                    self._visited_states.add(
                        state_after
                    )

                    self.stats[
                        "new_dom_states"
                    ] += 1

                    self._result[
                        "dom_states"
                    ].append(
                        {
                            "fingerprint":
                                state_after,

                            "url":
                                url_after,
                        }
                    )

            # =========================================================
            # 9. REAL NAVIGATION OBSERVATION
            # =========================================================

            if url_changed:

                self.stats[
                    "navigation_changes"
                ] += 1

                self._record_route(
                    url_after
                )

            # =========================================================
            # 10. CONVERGENCE
            # =========================================================

            if (
                state_changed
                or
                url_changed
                or
                evidence_changed
            ):

                self._no_change_rounds = 0

                log_info(
                    (
                        "Browser state/evidence changed | "
                        f"state_changed={state_changed} | "
                        f"url_changed={url_changed} | "
                        f"new_evidence={evidence_changed}"
                    )
                )

            else:

                self._no_change_rounds += 1

                log_info(
                    (
                        "Browser state converged for this cycle | "
                        f"stable={self._no_change_rounds}/"
                        f"{self.convergence_threshold}"
                    )
                )

            # =========================================================
            # 11. RUNTIME EVIDENCE MAY HAVE CREATED NEW SURFACES
            # =========================================================

            try:

                runtime = (
                    await self._read_runtime_export(
                        page
                    )
                )

                self._record_runtime_observation(
                    runtime
                )

                runtime_artifacts = (
                    self._extract_runtime_artifacts(
                        runtime
                    )
                )

                if runtime_artifacts:

                    await self.discover_surfaces(
                        page,
                        runtime_artifacts,
                    )

            except Exception as exc:

                log_debug(
                    (
                        "Final runtime observation failed: "
                        f"{type(exc).__name__}: {exc}"
                    )
                )

            # =========================================================
            # 12. STOP ONLY AFTER REAL CONVERGENCE
            # =========================================================

            if (
                self._no_change_rounds
                >= self.convergence_threshold
                and
                not self.has_work()
            ):

                log_info(
                    (
                        "Behavioral exploration converged naturally | "
                        f"cycles={self.stats['rounds']} | "
                        f"actions={self.stats['actions']} | "
                        f"states={self.stats['new_dom_states']} | "
                        f"routes={self.stats['new_routes']}"
                    )
                )

                break

        # -------------------------------------------------------------
        # Final observation
        # -------------------------------------------------------------

        try:

            await self._final_observation(
                page
            )

        except Exception as exc:

            log_debug(
                (
                    "Final behavioral observation failed: "
                    f"{type(exc).__name__}: {exc}"
                )
            )

        self._result[
            "stats"
        ] = dict(
            self.stats
        )

        return self._result
    
    # =========================================================================
    # LIVE DOM DISCOVERY
    # =========================================================================

    async def discover_live_dom_surfaces(
        self,
        page,
    ) -> None:

        if page is None:
            return

        try:

            if page.is_closed():
                return

        except Exception:

            return

        # ------------------------------------------------------------------
        # Forms
        # ------------------------------------------------------------------

        await self._queue_selector(
            page,
            "form",
            "form",
            115,
            "rendered-dom",
        )

        # ------------------------------------------------------------------
        # Editable controls
        # ------------------------------------------------------------------

        await self._queue_selector(
            page,
            (
                "input:not([type='hidden']):"
                "not([type='submit']):"
                "not([type='button']):"
                "not([type='reset']):"
                "not([type='file']):"
                "not([disabled]),"
                "textarea:not([disabled]),"
                "select:not([disabled]),"
                "[contenteditable='true']"
            ),
            "control",
            105,
            "rendered-dom",
        )

        # ------------------------------------------------------------------
        # Buttons / dynamic controls
        # ------------------------------------------------------------------

        await self._queue_selector(
            page,
            (
                "button:not([disabled]),"
                "[role='button'],"
                "[aria-expanded],"
                "[aria-haspopup],"
                "[data-toggle],"
                "[data-menu],"
                "[data-action]"
            ),
            "click",
            100,
            "rendered-dom",
        )

        # ------------------------------------------------------------------
        # Actual navigable links
        # ------------------------------------------------------------------

        await self._queue_selector(
            page,
            (
                "a[href],"
                "area[href]"
            ),
            "click",
            90,
            "rendered-dom",
        )

        # ------------------------------------------------------------------
        # Tabs
        # ------------------------------------------------------------------

        await self._queue_selector(
            page,
            (
                "[role='tab'],"
                "[role='tablist'] [aria-controls]"
            ),
            "click",
            95,
            "rendered-dom",
        )

        # ------------------------------------------------------------------
        # Menus / expandable navigation
        # ------------------------------------------------------------------

        await self._queue_selector(
            page,
            (
                "summary,"
                "[aria-expanded='false'],"
                "[aria-haspopup='true']"
            ),
            "click",
            92,
            "rendered-dom",
        )

        # ------------------------------------------------------------------
        # Hover surfaces
        # ------------------------------------------------------------------

        await self._queue_selector(
            page,
            (
                "[title],"
                "[data-tooltip],"
                "[aria-haspopup='true']"
            ),
            "hover",
            60,
            "rendered-dom",
        )

    # =========================================================================
    # QUEUE SELECTOR
    # =========================================================================

    async def _queue_selector(
        self,
        page,
        selector,
        action,
        priority,
        source,
    ):
        """
        Discover CURRENT live DOM elements.

        Important:
        - Elements come only from the currently rendered browser DOM.
        - No URL guessing.
        - No endpoint generation.
        - No wordlists.
        - No payloads.
        - No arbitrary HTTP requests.

        Each surface receives a browser-observed identity so that
        dynamically changing DOM does not depend entirely on a stale
        nth-index.
        """

        if page is None:
            return

        try:
            if page.is_closed():
                return
        except Exception:
            return

        try:
            locator = page.locator(selector)
            count = await locator.count()

        except Exception:
            return

        if count <= 0:
            return

        for index in range(min(count, 250)):

            try:
                element = locator.nth(index)

                if not await element.is_visible():
                    continue

            except Exception:
                continue

            # ------------------------------------------------------
            # Extract identity from the REAL rendered element
            # ------------------------------------------------------

            try:

                identity = await element.evaluate(
                    """
                    (e) => ({
                        tag: e.tagName.toLowerCase(),
                        id: e.id || "",
                        name: e.getAttribute("name") || "",
                        role: e.getAttribute("role") || "",
                        type: e.getAttribute("type") || "",
                        href: e.getAttribute("href") || "",
                        aria: e.getAttribute("aria-label") || "",
                        text: (
                            e.innerText ||
                            e.textContent ||
                            ""
                        ).trim().slice(0, 160),

                        data_testid:
                            e.getAttribute("data-testid") || "",

                        data_cy:
                            e.getAttribute("data-cy") || "",

                        title:
                            e.getAttribute("title") || ""
                    })
                    """
                )

            except Exception:
                continue

            # ------------------------------------------------------
            # Stable identity
            # ------------------------------------------------------

            identity_text = "|".join(
                str(identity.get(key, "") or "")
                for key in (
                    "tag",
                    "id",
                    "name",
                    "role",
                    "type",
                    "href",
                    "aria",
                    "text",
                    "data_testid",
                    "data_cy",
                    "title",
                )
            )

            identity_hash = hashlib.sha256(
                identity_text.encode(
                    "utf-8",
                    errors="ignore",
                )
            ).hexdigest()[:16]

            # ------------------------------------------------------
            # Do not permanently identify an element only by nth.
            #
            # nth is retained as a fallback locator because the DOM
            # can change after another interaction.
            # ------------------------------------------------------

            locator_expression = (
                f"{selector} >> nth={index}"
            )

            key = (
                action,
                source,
                identity_hash,
                locator_expression,
            )

            if key in self._seen_surfaces:
                continue

                        # The browser state is part of the interaction identity.
            #
            # This allows the same DOM control to become a new
            # interaction opportunity after a real SPA/DOM mutation.
            #
            # The fingerprint itself is based only on meaningful
            # rendered application controls.

            try:

                state = await self._state_fingerprint(
                    page
                )

            except Exception:

                state = ""

            surface = InteractionSurface(
                priority=int(priority),
                action=action,
                locator=locator_expression,
                source=source,
                metadata={
                    "index": index,
                    "selector": selector,
                    "identity": identity,
                    "identity_hash": identity_hash,
                    "state": state,
                },
            )

            await self.add_surface(
                surface
            )

    # =========================================================================
    # SAFE CLICK
    # =========================================================================

    async def _is_safe_click_target(
        self,
        element,
        identity=None,
    ) -> bool:
        """
        Decide whether a REAL rendered browser element may be activated.

        Phase-1 interaction is behavioral and target-scoped.

        Rules:

        - Only elements actually present in the rendered DOM are considered.
        - No predefined application paths are used.
        - No endpoint wordlists are used.
        - No synthetic URLs are generated.
        - Same-origin navigations may be interacted with.
        - External origins are observed but are NOT followed.
        - Non-web schemes are not activated.
        - Destructive/business-sensitive controls are observed but not executed.
        """

        if element is None:
            return False

        # -------------------------------------------------------------
        # Obtain identity from the REAL rendered element
        # -------------------------------------------------------------

        if identity is None:

            try:

                identity = await element.evaluate(
                    """
                    e => ({
                        tag: e.tagName || "",
                        type: e.getAttribute("type") || "",
                        text: (
                            e.innerText ||
                            e.textContent ||
                            ""
                        ).trim(),
                        aria: e.getAttribute("aria-label") || "",
                        name: e.getAttribute("name") || "",
                        id: e.id || "",
                        href: e.getAttribute("href") || ""
                    })
                    """
                )

            except Exception:

                return False

        if not isinstance(identity, dict):
            return False

        tag = str(
            identity.get(
                "tag",
                ""
            )
        ).lower()

        input_type = str(
            identity.get(
                "type",
                ""
            )
        ).lower()

        # -------------------------------------------------------------
        # Never activate submit/reset controls during discovery.
        #
        # Their presence remains an observation for later analysis.
        # -------------------------------------------------------------

        if input_type in (
            "submit",
            "reset",
        ):

            return False

        # -------------------------------------------------------------
        # ANCHOR SAFETY
        #
        # This is dynamic.
        #
        # We do NOT compare against a list such as:
        #
        # /login
        # /search
        # /admin
        # /products
        #
        # Instead we resolve the REAL href against the REAL browser
        # URL and compare origins.
        # -------------------------------------------------------------

        if tag == "a":

            href = str(
                identity.get(
                    "href",
                    ""
                ) or ""
            ).strip()

            if not href:
                return False

            try:

                from urllib.parse import (
                    urljoin,
                    urlparse,
                )

                current_url = str(
                    getattr(
                        element.page,
                        "url",
                        ""
                    ) or ""
                ).strip()

                if not current_url:
                    return False

                resolved = urljoin(
                    current_url,
                    href,
                )

                parsed_current = urlparse(
                    current_url
                )

                parsed_target = urlparse(
                    resolved
                )

                # -----------------------------------------------------
                # Ignore non-web browser schemes.
                #
                # These are not application navigations:
                #
                # javascript:
                # mailto:
                # tel:
                # data:
                # blob:
                # file:
                # -----------------------------------------------------

                if parsed_target.scheme.lower() not in (
                    "http",
                    "https",
                ):

                    return False

                # -----------------------------------------------------
                # REAL browser-origin comparison.
                #
                # Only the currently observed application origin may
                # be followed.
                # -----------------------------------------------------

                current_origin = (
                    parsed_current.scheme.lower(),
                    parsed_current.hostname.lower()
                    if parsed_current.hostname
                    else "",
                    parsed_current.port
                    or (
                        443
                        if parsed_current.scheme.lower() == "https"
                        else 80
                    ),
                )

                target_origin = (
                    parsed_target.scheme.lower(),
                    parsed_target.hostname.lower()
                    if parsed_target.hostname
                    else "",
                    parsed_target.port
                    or (
                        443
                        if parsed_target.scheme.lower() == "https"
                        else 80
                    ),
                )

                if current_origin != target_origin:

                    return False

            except Exception:

                return False

            return True

        # -------------------------------------------------------------
        # Destructive/business-sensitive controls.
        #
        # They are discovered and recorded but not activated.
        # -------------------------------------------------------------

        text = " ".join(
            str(
                identity.get(
                    key,
                    ""
                ) or ""
            ).lower()
            for key in (
                "text",
                "aria",
                "name",
                "id",
            )
        )

        blocked = (
            "delete",
            "remove",
            "destroy",
            "logout",
            "log out",
            "sign out",
            "signout",
            "purchase",
            "buy now",
            "checkout",
            "pay now",
            "payment",
            "transfer",
            "send money",
            "place order",
            "confirm order",
            "unsubscribe",
            "cancel subscription",
        )

        if any(
            token in text
            for token in blocked
        ):

            return False

        return True
   
    # =========================================================================
    # EXECUTION
    # =========================================================================

    async def _capture_state(
        self,
        page,
    ) -> dict[str, Any]:
        """
        Capture the currently reached browser state.

        This is an observation snapshot.

        It contains only state actually exposed by the browser:
            - URL
            - rendered DOM
            - visible interaction surface count
            - runtime observations

        It does not create crawl targets.
        """

        if page is None:
            return {}

        try:
            if page.is_closed():
                return {}
        except Exception:
            return {}

        try:
            url = str(
                page.url or ""
            )
        except Exception:
            url = ""

        try:
            dom = await page.locator(
                "body"
            ).inner_text(
                timeout=2000
            )
        except Exception:
            dom = ""

        runtime = await self._read_runtime_export(
            page
        )

        return {
            "url": url,
            "dom": dom[:20000],
            "runtime": runtime,
        }


    async def _state_fingerprint(
        self,
        page,
    ) -> str:
        """
        Return a deterministic fingerprint of the currently reached
        browser state.

        Two identical browser states produce the same fingerprint.
        """

        state = await self._capture_state(
            page
        )

        try:
            serialized = json.dumps(
                state,
                sort_keys=True,
                default=str,
                separators=(
                    ",",
                    ":",
                ),
            )
        except Exception:
            serialized = repr(
                state
            )

        return hashlib.sha256(
            serialized.encode(
                "utf-8",
                errors="replace",
            )
        ).hexdigest()

    # =========================================================================
    # CONTROL
    # =========================================================================

    async def _handle_control(
        self,
        locator,
    ) -> dict[str, Any]:

        try:

            tag = (
                await locator.evaluate(
                    "(e) => e.tagName"
                )
            ).lower()

        except Exception:

            return {
                "success": False,
                "reason": "control-detached",
            }

        # ------------------------------------------------------------------
        # SELECT
        # ------------------------------------------------------------------

        if tag == "select":

            try:

                options = await locator.locator(
                    "option"
                ).all()

                if not options:

                    return {
                        "success": False,
                        "reason": "no-options",
                    }

                # Selecting the first existing option is an observation
                # action only; no arbitrary attack value is introduced.

                value = await options[
                    0
                ].get_attribute(
                    "value"
                )

                if value is None:

                    return {
                        "success": False,
                        "reason": "option-has-no-value",
                    }

                await locator.select_option(
                    value
                )

                self.stats[
                    "controls_filled"
                ] += 1

                return {
                    "success": True,
                    "action": "select",
                    "value_observed": True,
                }

            except Exception as exc:

                return {
                    "success": False,
                    "reason": (
                        f"{type(exc).__name__}: "
                        f"{exc}"
                    ),
                }

        # ------------------------------------------------------------------
        # Checkbox / radio
        # ------------------------------------------------------------------

        try:

            input_type = (
                await locator.get_attribute(
                    "type"
                )
                or ""
            ).lower()

        except Exception:

            input_type = ""

        if input_type in (
            "checkbox",
            "radio",
        ):

            try:

                if not await locator.is_checked():

                    await locator.check()

                    self.stats[
                        "controls_filled"
                    ] += 1

                return {
                    "success": True,
                    "action": input_type,
                }

            except Exception as exc:

                return {
                    "success": False,
                    "reason": (
                        f"{type(exc).__name__}: "
                        f"{exc}"
                    ),
                }

        # ------------------------------------------------------------------
        # Sensitive controls
        # ------------------------------------------------------------------

        if input_type in (
            "password",
            "file",
        ):

            # Phase-1 records these controls but never invents
            # credentials or uploads a file.

            return {
                "success": True,
                "action": "observe-sensitive-control",
            }

        # ------------------------------------------------------------------
        # Editable text controls
        # ------------------------------------------------------------------
        #
        # Phase 1 is observation-only.
        #
        # Do NOT inject synthetic values into application inputs.
        # A synthetic value could change application state, submit data,
        # create artificial requests, or accidentally enter an application
        # workflow.
        #
        # The control itself is still valuable Phase-1 evidence because
        # Phase 2 can later decide whether and how it should be validated.
        # ------------------------------------------------------------------

        try:

            editable = await locator.is_editable()

        except Exception:

            editable = False

        if not editable:

            return {
                "success": False,
                "reason": "not-editable",
            }

        try:

            metadata = await locator.evaluate(
                """
                (e) => ({
                    tag: e.tagName.toLowerCase(),
                    name: e.getAttribute("name") || "",
                    type: e.getAttribute("type") || "text",
                    id: e.id || "",
                    placeholder:
                        e.getAttribute("placeholder") || "",
                    form:
                        e.form ? {
                            action: e.form.getAttribute("action") || "",
                            method:
                                (
                                    e.form.getAttribute("method")
                                    || "GET"
                                ).toUpperCase()
                        } : null
                })
                """
            )

        except Exception:

            metadata = {}

        self.stats[
            "controls_observed"
        ] = (
            self.stats.get(
                "controls_observed",
                0,
            )
            + 1
        )

        return {
            "success": True,
            "action": "observe-control",
            "observed_only": True,
            "metadata": metadata,
        }
    # =========================================================================
    # FORM OBSERVATION
    # =========================================================================

    async def _observe_form(
        self,
        locator,
    ) -> dict[str, Any]:

        self.stats[
            "forms"
        ] += 1

        try:

            action = (
                await locator.get_attribute(
                    "action"
                )
                or ""
            )

            method = (
                await locator.get_attribute(
                    "method"
                )
                or "GET"
            ).upper()

            controls = await locator.locator(
                "input,textarea,select,button"
            ).count()

            record = {
                "action": action,

                "method": method,

                "control_count": controls,
            }

            self._result[
                "forms"
            ].append(
                record
            )

            self.stats[
                "new_forms"
            ] += 1

            return {
                "success": True,

                "action": "observe-form",

                "form": record,
            }

        except Exception as exc:

            return {
                "success": False,

                "reason": (
                    f"{type(exc).__name__}: "
                    f"{exc}"
                ),
            }

    # CONTROL RECORD
    # =========================================================================

    def _record_control(
        self,
        surface,
        page,
    ) -> None:

        record = {
            "locator": surface.locator,

            "url": self._safe_page_url(
                page
            ),

            "source": surface.source,

            "metadata": dict(
                surface.metadata
            ),
        }

        # Avoid duplicate records.

        key = self._stable_metadata_key(
            record
        )

        existing = {
            self._stable_metadata_key(
                value
            )
            for value in self._result[
                "controls"
            ]
        }

        if key not in existing:

            self._result[
                "controls"
            ].append(
                record
            )

            self.stats[
                "new_controls"
            ] += 1

    # =========================================================================
    # RUNTIME ARTIFACT DISCOVERY
    # =========================================================================

    async def discover_surfaces(
        self,
        page,
        artifacts,
    ):
    
        if not isinstance(
            artifacts,
            (list, tuple),
        ):
            return

        values = self._as_list(
            artifacts
        )

        if not values:
            return

        # Runtime artifacts are interpreted as evidence that the
        # corresponding browser surface exists. We still query the
        # CURRENT DOM before interacting.

        rules = {

            "form": (
                "form",
                "form",
                115,
            ),

            "dynamic_ui": (
                (
                    "button,"
                    "[role='button'],"
                    "[aria-expanded],"
                    "[aria-haspopup]"
                ),
                "click",
                100,
            ),

            "dialog": (
                (
                    "dialog,"
                    "[role='dialog']"
                ),
                "dialog",
                98,
            ),

            "interactive_surface": (
                (
                    "button,"
                    "a[href],"
                    "[role='button'],"
                    "[role='tab']"
                ),
                "click",
                90,
            ),
        }

        for artifact in values:

            if not isinstance(
                artifact,
                dict,
            ):

                category = getattr(
                    artifact,
                    "category",
                    None,
                )

                if category is None:

                    category = getattr(
                        artifact,
                        "type",
                        None,
                    )

            else:

                category = (
                    artifact.get(
                        "category"
                    )
                    or
                    artifact.get(
                        "type"
                    )
                )

            rule = rules.get(
                category
            )

            if rule is None:
                continue

            selector, action, priority = rule

            await self._queue_selector(
                page,
                selector,
                action,
                priority,
                "runtime-observation",
            )


            
    async def _collect_post_action_runtime(
        self,
        page,
    ) -> None:
        """
        Collect browser-runtime evidence produced after a real interaction.

        The browser has already performed the action.

        We only read:
            - navigation events
            - fetch/XHR observations
            - other runtime events

        No request is generated here.
        """

        runtime = await self._read_runtime_export(
            page
        )

        self._record_runtime_observation(
            runtime
        )

        if not isinstance(
            runtime,
            dict,
        ):
            return

        events = self._as_list(
            runtime.get(
                "events",
                []
            )
        )

        for event in events:

            if not isinstance(
                event,
                dict,
            ):
                continue

            category = str(
                event.get(
                    "category"
                )
                or ""
            ).lower()

            target = str(
                event.get(
                    "target"
                )
                or ""
            ).strip()

            metadata = event.get(
                "metadata"
            )

            if not isinstance(
                metadata,
                dict,
            ):
                metadata = {}

            # ---------------------------------------------------------
            # REAL browser navigation
            # ---------------------------------------------------------

            if (
                category == "navigation"
                and target
            ):

                self._record_route(
                    target
                )

            # ---------------------------------------------------------
            # REAL browser network activity
            # ---------------------------------------------------------

            elif (
                category == "network"
                and target
            ):

                key = (
                    str(
                        event.get(
                            "subtype"
                        )
                        or ""
                    ),

                    str(
                        metadata.get(
                            "method"
                        )
                        or "GET"
                    ).upper(),

                    target,
                )

                if key in self._observed_network:
                    continue

                self._observed_network.add(
                    key
                )

                self._result[
                    "network_observations"
                ].append(
                    {
                        "type":
                            event.get(
                                "subtype"
                            )
                            or "network",

                        "url":
                            target,

                        "method":
                            metadata.get(
                                "method",
                                "GET",
                            ),

                        "status":
                            metadata.get(
                                "status"
                            ),

                        "metadata":
                            metadata,
                    }
                )

                self.stats[
                    "network_observations"
                ] += 1

        
    # =========================================================================
    # RUNTIME EXPORT
    # =========================================================================

    async def _read_runtime_export(
        self,
        page,
    ) -> dict[str, Any]:
        """
        Read the runtime export produced by the browser instrumentation.

        Observation-only.

        This method never:
            - generates URLs
            - generates parameter names
            - sends HTTP requests
            - fuzzes inputs
            - creates attack payloads
        """

        if page is None:
            return {}

        try:
            if page.is_closed():
                return {}
        except Exception:
            return {}

        try:
            runtime = await page.evaluate(
                """
                () => {
                    if (
                        window.__AVAF__ &&
                        typeof window.__AVAF__.export === "function"
                    ) {
                        return window.__AVAF__.export();
                    }

                    return {};
                }
                """
            )
        except Exception as exc:
            logger.debug(
                "RUNTIME",
                (
                    "Runtime export read failed: "
                    f"{type(exc).__name__}: {exc}"
                ),
            )
            return {}

        if not isinstance(runtime, dict):
            return {}

        return runtime


    async def _read_runtime_artifacts(
        self,
        page,
    ) -> list[dict[str, Any]]:
        """
        Read artifacts already observed by the browser runtime.

        This is strictly observation-only.
        """

        runtime = await self._read_runtime_export(
            page
        )

        if not runtime:
            return []

        self._record_runtime_observation(
            runtime
        )

        return self._extract_runtime_artifacts(
            runtime
        )

    # =========================================================================
    # RUNTIME ARTIFACTS
    # =========================================================================

    def _extract_runtime_artifacts(
        self,
        runtime,
    ) -> list:

        if not isinstance(
            runtime,
            dict,
        ):

            return []

        artifacts = runtime.get(
            "artifacts",
            {}
        )

        if not isinstance(
            artifacts,
            dict,
        ):

            return []

        result = []

        for value in artifacts.values():

            result.extend(
                self._as_list(
                    value
                )
            )

        return result

    # =========================================================================
    # RUNTIME OBSERVATION
    # =========================================================================

    def _record_runtime_observation(
        self,
        runtime,
    ) -> None:

        if not isinstance(
            runtime,
            dict,
        ):

            return

        events = self._as_list(
            runtime.get(
                "events",
                []
            )
        )

        for event in events:

            self._result[
                "runtime_observations"
            ].append(
                event
            )

            self.stats[
                "runtime_events"
            ] += 1

    # =========================================================================
    # POST-ACTION RUNTIME CALLBACK
    # =========================================================================

    async def _runtime_feedback(
        self,
        page,
        event,
    ) -> None:

        if self.runtime_callback is None:
            return

        try:

            result = self.runtime_callback(
                page,
                event,
            )

            if asyncio.iscoroutine(
                result
            ):

                await result

            self.stats[
                "runtime_feedback"
            ] += 1

        except Exception:

            # Callback failure must never kill discovery.

            return
        
    # =========================================================================
    # STATE FINGERPRINT
    # =========================================================================

    async def _state_fingerprint(
        self,
        page,
    ) -> str:
        """
        Fingerprint the meaningful rendered browser state.

        The fingerprint is used for behavioral convergence only.

        It does NOT:
            - generate routes
            - generate parameters
            - infer vulnerabilities
            - replace network evidence

        The state contains:
            - current browser URL
            - page title
            - visible interaction surface identity
            - visible form/control identity

        This deliberately ignores volatile full-page HTML so that
        harmless dynamic noise does not keep exploration alive forever.
        """

        if page is None:
            return ""

        try:

            if page.is_closed():
                return ""

        except Exception:

            return ""

        try:

            state = await page.evaluate(
                """
                () => {

                    const visible = (el) => {

                        if (!el) {
                            return false;
                        }

                        const style =
                            window.getComputedStyle(el);

                        const rect =
                            el.getBoundingClientRect();

                        return (
                            style.display !== "none" &&
                            style.visibility !== "hidden" &&
                            rect.width > 0 &&
                            rect.height > 0
                        );
                    };

                    const normalize = (value) => {

                        return String(
                            value || ""
                        )
                        .replace(/\\s+/g, " ")
                        .trim()
                        .slice(0, 300);
                    };

                    const elements = [];

                    const nodes = document.querySelectorAll(
                        [
                            "a[href]",
                            "button",
                            "[role='button']",
                            "[role='tab']",
                            "[aria-expanded]",
                            "[aria-haspopup]",
                            "input:not([type='hidden'])",
                            "textarea",
                            "select",
                            "[contenteditable='true']",
                            "form"
                        ].join(",")
                    );

                    for (const el of nodes) {

                        if (!visible(el)) {
                            continue;
                        }

                        const rect =
                            el.getBoundingClientRect();

                        elements.push({
                            tag:
                                el.tagName || "",

                            type:
                                el.getAttribute("type") || "",

                            id:
                                el.id || "",

                            name:
                                el.getAttribute("name") || "",

                            role:
                                el.getAttribute("role") || "",

                            aria:
                                el.getAttribute("aria-label") || "",

                            text:
                                normalize(
                                    el.innerText ||
                                    el.textContent ||
                                    ""
                                ),

                            href:
                                el.getAttribute("href") || "",

                            action:
                                el.getAttribute("action") || "",

                            expanded:
                                el.getAttribute(
                                    "aria-expanded"
                                ) || "",

                            x:
                                Math.round(rect.x),

                            y:
                                Math.round(rect.y),

                            width:
                                Math.round(rect.width),

                            height:
                                Math.round(rect.height)
                        });
                    }

                    return {
                        url:
                            location.href || "",

                        title:
                            document.title || "",

                        elements
                    };
                }
                """
            )

        except Exception:

            return ""

        try:

            serialized = json.dumps(
                state,
                sort_keys=True,
                separators=(
                    ",",
                    ":",
                ),
                ensure_ascii=False,
            )

        except Exception:

            serialized = repr(
                state
            )

        return hashlib.sha256(
            serialized.encode(
                "utf-8",
                errors="ignore",
            )
        ).hexdigest()

    # =========================================================================
    # STATE CAPTURE
    # =========================================================================

    async def _capture_state(
        self,
        page,
    ):

        try:

            if page.is_closed():
                return None

        except Exception:

            return None

        try:

            url = page.url or ""

        except Exception:

            url = ""

        try:

            html = await page.locator(
                "html"
            ).evaluate(
                "(e) => e.outerHTML"
            )

        except Exception:

            try:

                html = await page.content()

            except Exception:

                html = ""

        if not html:

            return None

        try:

            title = await page.title()

        except Exception:

            title = ""

        fingerprint = hashlib.sha256(
            (
                url
                + "\n"
                + html
            ).encode(
                "utf-8",
                errors="ignore",
            )
        ).hexdigest()

        return {
            "fingerprint": fingerprint,

            "url": url,

            "title": title,

            "html_size": len(html),
        }

    # =========================================================================
    # SETTLE
    # =========================================================================

    async def _settle(
        self,
        page,
        milliseconds=None,
    ) -> None:

        if milliseconds is None:

            milliseconds = (
                self.state_wait
            )

        try:

            await page.wait_for_timeout(
                milliseconds
            )

        except Exception:

            return

    async def _settle_after_action(
        self,
        page,
        before_url,
    ) -> None:

        await self._settle(
            page,
            self.state_wait,
        )

        after_url = self._safe_page_url(
            page
        )

        if (
            after_url
            and
            after_url != before_url
        ):

            await self._settle(
                page,
                self.post_navigation_wait,
            )

    # =========================================================================
    # ROUTE RECORDING
    # =========================================================================

    def _record_route(
        self,
        url: str,
    ) -> None:

        if not url:
            return

        if url in self._observed_urls:
            return

        self._observed_urls.add(
            url
        )

        self._result[
            "navigations"
        ].append(
            url
        )

        self._result[
            "routes"
        ].append(
            url
        )

        self.stats[
            "new_routes"
        ] += 1

    # =========================================================================
    # FINAL OBSERVATION
    # =========================================================================

    async def _final_observation(
        self,
        page,
    ) -> None:

        if page is None:
            return

        try:

            await self._settle(
                page,
                self.state_wait,
            )

        except Exception:

            pass

        await self.discover_live_dom_surfaces(
            page
        )

        runtime = await self._read_runtime_export(
            page
        )

        self._record_runtime_observation(
            runtime
        )

        state = await self._capture_state(
            page
        )

        if state:

            fingerprint = state[
                "fingerprint"
            ]

            if fingerprint not in self._visited_states:

                self._visited_states.add(
                    fingerprint
                )

                self.stats[
                    "new_dom_states"
                ] += 1

                self._result[
                    "dom_states"
                ].append(
                    state
                )

    # =========================================================================
    # SAFE PAGE URL
    # =========================================================================

    @staticmethod
    def _safe_page_url(
        page,
    ) -> str:

        try:

            return page.url or ""

        except Exception:

            return ""
